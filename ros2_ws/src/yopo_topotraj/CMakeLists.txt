cmake_minimum_required(VERSION 3.16)
project(yopo_topotraj)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(message_filters REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
find_package(OpenCV REQUIRED)
find_package(NLopt REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
include_directories(include vendor/plan_env/include vendor/path_searching/include vendor/bspline_opt/include vendor/plan_manage/include)
add_library(topotraj_map vendor/plan_env/src/sdf_map.cpp vendor/plan_env/src/raycast.cpp vendor/plan_env/src/edt_environment.cpp)
target_compile_features(topotraj_map PUBLIC cxx_std_17)
target_include_directories(topotraj_map PUBLIC ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(topotraj_map Eigen3::Eigen ${PCL_LIBRARIES} ${OpenCV_LIBRARIES})
ament_target_dependencies(topotraj_map rclcpp sensor_msgs geometry_msgs nav_msgs visualization_msgs cv_bridge message_filters pcl_conversions)
add_library(topotraj_search vendor/path_searching/src/topo_prm.cpp vendor/path_searching/src/astar.cpp vendor/path_searching/src/kinodynamic_astar.cpp)
target_link_libraries(topotraj_search topotraj_map)
add_library(topotraj_optimizer vendor/bspline_opt/src/bspline_optimizer.cpp)
target_link_libraries(topotraj_optimizer topotraj_map nlopt)
add_library(topotraj_bspline vendor/bspline/src/non_uniform_bspline.cpp)
target_compile_features(topotraj_bspline PUBLIC cxx_std_17)
target_include_directories(topotraj_bspline PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/vendor/bspline/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(topotraj_bspline Eigen3::Eigen)
ament_target_dependencies(topotraj_bspline rclcpp)
add_library(topotraj_poly vendor/poly_traj/src/polynomial_traj.cpp)
target_compile_features(topotraj_poly PUBLIC cxx_std_17)
target_include_directories(topotraj_poly PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/vendor/poly_traj/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(topotraj_poly PUBLIC Eigen3::Eigen)
add_library(topotraj_manager vendor/plan_manage/src/planner_manager.cpp)
target_link_libraries(topotraj_manager topotraj_bspline topotraj_poly topotraj_search topotraj_optimizer)
add_executable(check_spline src/check_spline.cpp)
target_link_libraries(check_spline topotraj_bspline topotraj_poly)
add_executable(check_components src/check_components.cpp)
target_link_libraries(check_components topotraj_search topotraj_optimizer)
add_executable(check_online_mapping src/check_online_mapping.cpp)
target_link_libraries(check_online_mapping topotraj_map)
add_executable(topotraj_node src/topotraj_node.cpp vendor/plan_manage/src/topo_replan_fsm.cpp)
target_link_libraries(topotraj_node topotraj_manager)
ament_target_dependencies(topotraj_node trajectory_msgs std_msgs)
add_executable(topotraj_runtime src/topotraj_runtime.cpp vendor/plan_manage/src/topo_replan_fsm.cpp)
target_link_libraries(topotraj_runtime topotraj_manager yaml-cpp)
ament_target_dependencies(topotraj_runtime trajectory_msgs std_msgs rosgraph_msgs)
install(TARGETS topotraj_bspline topotraj_poly topotraj_map topotraj_search topotraj_optimizer topotraj_manager EXPORT topotraj_components DESTINATION lib)
install(TARGETS check_spline check_components check_online_mapping topotraj_node topotraj_runtime DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY vendor/bspline/include/ vendor/poly_traj/include/ DESTINATION include)
install(DIRECTORY include/ vendor/plan_env/include/ vendor/path_searching/include/ vendor/bspline_opt/include/ vendor/plan_manage/include/ DESTINATION include)
ament_export_targets(topotraj_components HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp Eigen3)
ament_package()
