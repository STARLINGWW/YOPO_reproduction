cmake_minimum_required(VERSION 3.16)
project(yopo_math_reference)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
get_filename_component(REPRO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(UPSTREAM "${REPRO_ROOT}/external/audit_sources/main/flightlib")
add_executable(matrix_reference src/matrix_reference.cpp
  "${UPSTREAM}/src/grad_traj_optimization/qp_generator.cpp")
target_compile_features(matrix_reference PRIVATE cxx_std_17)
target_include_directories(matrix_reference PRIVATE "${UPSTREAM}/include")
target_link_libraries(matrix_reference Eigen3::Eigen)
add_executable(voxel_reference src/voxel_reference.cpp)
target_compile_features(voxel_reference PRIVATE cxx_std_17)
target_include_directories(voxel_reference PRIVATE "${UPSTREAM}/third_party/arc_utilities/include")
target_link_libraries(voxel_reference Eigen3::Eigen)
add_executable(cost_reference src/cost_reference.cpp
  "${UPSTREAM}/src/grad_traj_optimization/grad_traj_optimizer.cpp"
  "${UPSTREAM}/src/grad_traj_optimization/qp_generator.cpp")
target_compile_features(cost_reference PRIVATE cxx_std_17)
target_include_directories(cost_reference PRIVATE shims "${UPSTREAM}/include")
target_link_libraries(cost_reference Eigen3::Eigen yaml-cpp)
add_executable(sdf_reference src/sdf_reference.cpp)
target_compile_features(sdf_reference PRIVATE cxx_std_17)
target_compile_options(sdf_reference PRIVATE -O3)
target_include_directories(sdf_reference PRIVATE "${UPSTREAM}/third_party/arc_utilities/include")
target_link_libraries(sdf_reference Eigen3::Eigen)
install(TARGETS matrix_reference voxel_reference cost_reference sdf_reference DESTINATION lib/${PROJECT_NAME})
ament_package()
