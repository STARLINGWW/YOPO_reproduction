cmake_minimum_required(VERSION 3.16)
project(yopo_expert_core)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(NLopt REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(UPSTREAM "${ROOT}/external/audit_sources/main/flightlib")
add_library(yopo_expert SHARED src/expert.cpp "${UPSTREAM}/src/grad_traj_optimization/grad_traj_optimizer.cpp" "${UPSTREAM}/src/grad_traj_optimization/qp_generator.cpp")
target_compile_features(yopo_expert PUBLIC cxx_std_17)
target_include_directories(yopo_expert PRIVATE adapter "${UPSTREAM}/include" "${UPSTREAM}/third_party/arc_utilities/include")
target_link_libraries(yopo_expert Eigen3::Eigen yaml-cpp nlopt)
install(TARGETS yopo_expert DESTINATION lib)
ament_package()
