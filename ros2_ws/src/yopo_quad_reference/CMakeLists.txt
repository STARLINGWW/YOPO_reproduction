cmake_minimum_required(VERSION 3.16)
project(yopo_quad_reference)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(so3_reference SHARED src/reference.cpp vendor/SO3Control.cpp)
target_compile_features(so3_reference PUBLIC cxx_std_17)
target_include_directories(so3_reference PRIVATE vendor)
target_link_libraries(so3_reference Eigen3::Eigen)
install(TARGETS so3_reference DESTINATION lib)
ament_package()
